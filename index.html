        function avviaPartita() {

            const catScelta = document.getElementById('category').value;

            impostazioneTempo = parseInt(document.getElementById('timerSet').value);

            

            if (catScelta === 'all') {

                mazzoAttivo = [...carteTaboo];

            } else {

                mazzoAttivo = carteTaboo.filter(c => c.cat === catScelta);

            }


            for (let i = mazzoAttivo.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

            }


            indiceCorrente = 0;

            

            document.getElementById('setupPanel').style.display = 'none';

            document.getElementById('gameArea').style.display = 'flex';

            

            renderizzaCarta();

            avviaTimer();

        }


        function renderizzaCarta() {

            if(indiceCorrente >= mazzoAttivo.length) {

                clearInterval(timerId);

                alert("Ottimo lavoro! Avete completato le carte di questa sessione!");

                ritornaAlMenu();

                return;

            }


            const scheda = mazzoAttivo[indiceCorrente];

            const contenitoreCarta = document.getElementById('activeCard');

            

            contenitoreCarta.className = 'card theme-' + scheda.cat;

            

            document.getElementById('currentSubject').innerText = scheda.label;

            document.getElementById('targetWord').innerText = scheda.word;

            

            const listContainer = document.getElementById('forbiddenList');

            listContainer.innerHTML = '';

            

            scheda.forbidden.forEach(parolaProibita => {

                const elementoLista = document.createElement('li');

                elementoLista.innerText = parolaProibita;

                listContainer.appendChild(elementoLista);

            });

        }


        function avanzaMazzo() {

            indiceCorrente++;

            renderizzaCarta();

                avviaTimer(); 

            }

        }


        function avviaTimer() {

            clearInterval(timerId);

            if(impostazioneTempo === 0) {

                document.getElementById('timerContainer').style.display = 'none';

                return;

            }

            

            document.getElementById('timerContainer').style.display = 'flex';

            tempoRimanente = impostazioneTempo;

            document.getElementById('timerDisplay').innerText = tempoRimanente;

            

            timerId = setInterval(() => {

                tempoRimanente--;

                document.getElementById('timerDisplay').innerText = tempoRimanente;

                

                if(tempoRimanente <= 0) {

                    clearInterval(timerId);

                    alert("Tempo Scaduto!");

                    avanzaMazzo();

                }

            }, 1000);

        }


        function ritornaAlMenu() {

            clearInterval(timerId);

            document.getElementById('setupPanel').style.display = 'block';

            document.getElementById('gameArea').style.display = 'none';

        }
